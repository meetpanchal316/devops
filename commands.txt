services:
  backend:
    image: randomquotegeneratorbackend:V1
    build: 
      context: ./backend
      dockerfile: Dockerfile
    ports:
      - 5000:5000
    restart: unless-stopped
    container_name: backend_for_randquotegen
  frontend:
    image: randomquotegeneratorfrontend:V1
    depends_on:
      - backend
    build: 
      context: ./frontend
      dockerfile: Dockerfile

      - 3000:3000
    restart: unless-stopped
    container_name: frontend_for_randquotegen

  replicas: 1

      labels:

    spec:

          image: panchalmeet/bmicalbackend:v1

to start jenkins with docker 

jenkins dockerfile 
FROM jenkins/jenkins:lts-jdk17

USER root

# Install dependencies
RUN apt-get update && \
    apt-get install -y \
      docker.io \
      curl \
      ca-certificates \
      apt-transport-https \
      gnupg && \
    rm -rf /var/lib/apt/lists/*

# Install kubectl
RUN curl -fsSL https://dl.k8s.io/release/stable.txt | \
    xargs -I {} curl -fsSL https://dl.k8s.io/release/{}/bin/linux/amd64/kubectl -o /usr/local/bin/kubectl && \
    chmod +x /usr/local/bin/kubectl

# Install gcloud + GKE auth plugin
RUN curl -fsSL https://packages.cloud.google.com/apt/doc/apt-key.gpg | gpg --dearmor -o /usr/share/keyrings/cloud.google.gpg && \
    echo "deb [signed-by=/usr/share/keyrings/cloud.google.gpg] https://packages.cloud.google.com/apt cloud-sdk main" \
    > /etc/apt/sources.list.d/google-cloud-sdk.list && \
    apt-get update && \
    apt-get install -y google-cloud-sdk google-cloud-sdk-gke-gcloud-auth-plugin && \
    rm -rf /var/lib/apt/lists/*

USER jenkins


build this docker file 
docker build -t jenkins-with-docker .

run this command 
getent group docker

docker run -d \
  --name jenkins \
  -p 8081:8080 \
  -p 50000:50000 \
  -v jenkins_home:/var/jenkins_home \
  -v /var/run/docker.sock:/var/run/docker.sock \
  -v $HOME/.kube:/var/jenkins_home/.kube \
  --group-add 108 \

  --display-name="Jenkins CI Service Account"

gcloud projects add-iam-policy-binding $PROJECT_ID \
  --member="serviceAccount:jenkins-ci@$PROJECT_ID.iam.gserviceaccount.com" \
  --role="roles/container.developer"

gcloud iam service-accounts keys create jenkins-key.json \
  --iam-account=jenkins-ci@$PROJECT_ID.iam.gserviceaccount.com

Activate CI authentication (NON-INTERACTIVE)
gcloud auth activate-service-account \
  --key-file=jenkins-key.json
